
module Async_Counter(
    input clk, reset,
    output reg [2:0] Q
);
always @(negedge clk or posedge reset)
begin
    if(reset)
        Q[0] <= 0;
    else
        Q[0] <= ~Q[0];
end
always @(negedge Q[0] or posedge reset)
begin
    if(reset)
        Q[1] <= 0;
    else
        Q[1] <= ~Q[1];
end
always @(negedge Q[1] or posedge reset)
begin
    if(reset)
        Q[2] <= 0;
    else
        Q[2] <= ~Q[2];
end
endmodule
module Async_Counter_tb;
reg clk, reset;
wire [2:0] Q;
Async_Counter uut(clk, reset, Q);
initial begin
    clk = 0;
    forever #5 clk = ~clk;
end
initial begin
    reset = 1;
    #10 reset = 0;
    #160 reset = 1;
    #10 reset = 0;
    #50 $finish;
end
initial
    $monitor("Time=%0t Reset=%b Q=%b", $time, reset, Q);
endmodule
