
#include <reg51.h>

unsigned char seg_code[] = {
    0x3F,0x06,0x5B,0x4F,0x66,
    0x6D,0x7D,0x07,0x7F,0x6F,
    0x77,0x7C,0x39,0x5E,0x79,0x71
};

void delay()
{
    unsigned int i,j;
    for(i=0;i<10;i++)
        for(j=0;j<1275;j++);
}

void display(unsigned char num)
{
    P2 = 0x00;                      // Disable both digits
    P1 = seg_code[(num >> 4) & 0x0F]; // Upper nibble
    P2 = 0x01;                      // Select first digit
    delay();

    P2 = 0x00;                      // Disable both digits
    P1 = seg_code[num & 0x0F];      // Lower nibble
    P2 = 0x02;                      // Select second digit
    delay();
}

void main()
{
    unsigned int count;
    unsigned int i;

    while(1)
    {
        for(count=0;count<256;count++)
        {
            for(i=0;i<100;i++)
            {
                display((unsigned char)count);
            }
        }
    }
}
