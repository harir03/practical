
#include <LPC214x.h>

#define ROW 0x000F0000// Rows P1.16-P1.19
#define COL 0x00F00000// Columns P1.20-P1.23

unsigned char key[4][4]={
{'1','2','3','A'},
{'4','5','6','B'},
{'7','8','9','C'},
{'*','0','#','D'}
};// Keypad characters

void delay()// Generate delay
{
unsigned int i;
for(i=0; i<10000; i++);
}

unsigned char scan_key()// Scan keypad
{
unsigned int row, col;

for(row=0; row<4; row++)
{
IOSET1 = ROW;
IOCLR1 = (1<<(16+row));
delay();

for(col=0; col<4; col++)
{
if(!(IOPIN1 & (1<<(20+col))))
{
delay();
if(!(IOPIN1 & (1<<(20+col))))
return key[row][col];
}
}
}
return 0;
}

void main()
{
unsigned char key_value;

IODIR1 = (IODIR1 | ROW) & ~COL;// Configure rows and columns
IOSET1 = ROW;// Set all rows HIGH

while(1)
{
key_value = scan_key();// Read pressed key
if(key_value != 0)
{
while(scan_key() != 0);// Wait for key release
delay();
}
}
}
