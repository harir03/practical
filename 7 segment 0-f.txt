
#include <reg51.h>

unsigned char seg_code[] = {
    0x3F,0x06,0x5B,0x4F,0x66,
    0x6D,0x7D,0x07,0x7F,0x6F,
    0x77,0x7C,0x39,0x5E,0x79,0x71
};

void delay()
{
    unsigned int i,j;
    for(i=0;i<10;i++)
        for(j=0;j<1275;j++);
}

void main()
{
    unsigned char count;

    while(1)
    {
        for(count=0;count<16;count++)
        {
            P2 = 0x00;              // Disable digits
            P1 = seg_code[count];   // Send segment pattern
            P2 = 0x01;              // Select first digit
            delay();
        }
    }
}
