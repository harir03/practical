module JK_FF(
    input J, K, clk,
    output reg Q
);
always @(posedge clk)
begin
    case({J,K})
        2'b00: Q <= Q;
        2'b01: Q <= 0;
        2'b10: Q <= 1;
        2'b11: Q <= ~Q;
    endcase
end
endmodule
module JK_FF_tb;
reg J, K, clk;
wire Q;
JK_FF uut(J,K,clk,Q);
initial begin
    clk = 0;
    forever #5 clk = ~clk;
end
initial begin
    J=0; K=0; #10;
    J=0; K=1; #10;
    J=1; K=0; #10;
    J=1; K=1; #20;
    $finish;
end
initial
    $monitor("J=%b K=%b Q=%b",J,K,Q);
endmodule
module T_FF(
    input T, clk,
    output reg Q
);
always @(posedge clk)
begin
    if(T)
        Q <= ~Q;
    else
        Q <= Q;
end
endmodule
module T_FF_tb;
reg T, clk;
wire Q;
T_FF uut(T,clk,Q);
initial begin
    clk = 0;
    forever #5 clk = ~clk;
end
initial begin
    T=0; #20;
    T=1; #40;
    T=0; #20;
    $finish;
end
initial
    $monitor("T=%b Q=%b",T,Q);
endmodule
